CREATE DATABASE IF NOT EXISTS quickbite CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE quickbite;

-- Tabela de produtos (lanches, bebidas, sobremesas, acompanhamentos, combos)
CREATE TABLE produto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    categoria ENUM('LANCHE','BEBIDA','SOBREMESA','ACOMPANHAMENTO','COMBO') NOT NULL,
    preco DECIMAL(6,2) NOT NULL
);

-- Tabela de pedidos
CREATE TABLE pedido (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tipo_consumo ENUM('LOCAL','LEVAR') NOT NULL,
    forma_pagamento ENUM('PIX','CARTAO','DINHEIRO'),
    total DECIMAL(8,2),
    senha VARCHAR(10),
    status ENUM('ABERTO','CONFIRMADO','CANCELADO') DEFAULT 'ABERTO',
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Tabela de itens do pedido (relação pedido <-> produto)
CREATE TABLE item_pedido (
    id INT AUTO_INCREMENT PRIMARY KEY,
    pedido_id INT NOT NULL,
    produto_id INT NOT NULL,
    FOREIGN KEY (pedido_id) REFERENCES pedido(id) ON DELETE CASCADE,
    FOREIGN KEY (produto_id) REFERENCES produto(id)
);
